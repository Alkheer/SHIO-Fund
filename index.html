-- شغّل هذا الملف كاملاً في Supabase > SQL Editor create table if not exists profiles ( id uuid primary key references auth.users on delete cascade, full_name text, role text not null default 'region_manager' check (role in ('admin','region_manager')), region text ); alter table members add column if not exists region text, add column if not exists address text, add column if not exists photo_path text, add column if not exists created_by uuid default auth.uid(), add column if not exists created_at timestamptz default now(); create unique index if not exists members_number_key on members(member_number); create or replace function is_admin() returns boolean language sql security definer stable set search_path = public as $$ select exists(select 1 from profiles where id = auth.uid() and role = 'admin') $$; create or replace function my_region() returns text language sql security definer stable set search_path = public as $$ select region from profiles where id = auth.uid() $$; alter table profiles enable row level security; alter table members enable row level security; drop policy if exists "read for logged in" on members; drop policy if exists "insert for logged in" on members; create policy p_self on profiles for select to authenticated using (id = auth.uid() or is_admin()); create policy m_sel on members for select to authenticated using (is_admin() or region = my_region()); create policy m_ins on members for insert to authenticated with check (is_admin() or region = my_region()); create policy m_upd on members for update to authenticated using (is_admin() or region = my_region()) with check (is_admin() or region = my_region()); create policy m_del on members for delete to authenticated using (is_admin());
